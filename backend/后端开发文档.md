#### JWT配置 (jwt.js)

> 注意：`jwt.js` 导出的是**工具函数**而非配置对象；密钥与过期时间配置在 `src/config/index.js` 的 `config.jwt` 项（secret / expiresIn / refreshExpiresIn / adminExpiresIn）。
> 安全扫描 H-4 后，令牌一律携带用途声明 `type`，校验方按用途白名单接受；没有 `type` 的历史令牌判为无效。
> 安全扫描 M-2 后，管理端令牌另带 `adminScope`（`account`/`role`/`emergency`）声明主体类型，`adminAuth` 每次请求据此回读实时状态与权限；缺 `adminScope` 的历史管理端令牌判为无效，时效改用 `adminExpiresIn`（默认 `12h`）。

```javascript
// src/config/jwt.js
const jwt = require('jsonwebtoken')
const config = require('./index')

// 用途声明：签发与校验共用同一份常量，避免字符串拼写漂移
const TOKEN_TYPES = {
  ACCESS: 'access',
  REFRESH: 'refresh',
  ADMIN: 'admin',
  ADMIN_REFRESH: 'admin_refresh'
}
const TOKEN_TYPE_VALUES = Object.values(TOKEN_TYPES)
const REFRESH_LIKE_TYPES = [TOKEN_TYPES.REFRESH, TOKEN_TYPES.ADMIN_REFRESH]

// 未知/缺省 type 直接抛错：不给「忘记带 type」留活口
const assertTokenType = (type, signer) => {
  if (!TOKEN_TYPE_VALUES.includes(type)) {
    throw new Error(`${signer} 必须指定合法 type，收到: ${type}`)
  }
  return type
}

const generateToken = (payload, type = TOKEN_TYPES.ACCESS) => {
  assertTokenType(type, 'generateToken')
  const expiresIn = REFRESH_LIKE_TYPES.includes(type)
    ? config.jwt.refreshExpiresIn
    : config.jwt.expiresIn
  return jwt.sign({ ...payload, type }, config.jwt.secret, { expiresIn })
}

const generateTokenPair = (payload) => {
  return {
    accessToken: generateToken(payload, TOKEN_TYPES.ACCESS),
    refreshToken: generateToken(payload, TOKEN_TYPES.REFRESH),
    expiresIn: config.jwt.expiresIn,
    refreshExpiresIn: config.jwt.refreshExpiresIn
  }
}

// allowedTypes 省略时只验签（保持既有调用方语义）；所有鉴权入口都必须显式传白名单
const verifyToken = (token, allowedTypes = null) => {
  try {
    const decoded = jwt.verify(token, config.jwt.secret)
    if (allowedTypes && !allowedTypes.includes(decoded.type)) return null
    return decoded
  } catch (error) {
    return null // 校验失败返回 null，由调用方决定如何处理
  }
}

// 只接受 type=refresh，且重签只结转 userId（不复制旧 payload 的其他声明）
const refreshAccessToken = (refreshToken) => {
  const decoded = verifyToken(refreshToken, [TOKEN_TYPES.REFRESH])
  if (!decoded) throw new Error('无效的refresh token')
  return {
    accessToken: generateToken({ userId: decoded.userId }, TOKEN_TYPES.ACCESS),
    refreshToken: generateToken({ userId: decoded.userId }, TOKEN_TYPES.REFRESH),
    expiresIn: config.jwt.expiresIn
  }
}

// 管理端签发：expiresIn 由调用方给定，type 强制
const signToken = (payload, expiresIn, type) => {
  assertTokenType(type, 'signToken')
  return jwt.sign({ ...payload, type }, config.jwt.secret, { expiresIn })
}

// 另导出 decodeToken（不验签，仅解码）
module.exports = {
  TOKEN_TYPES,
  generateToken,
  generateTokenPair,
  verifyToken,
  refreshAccessToken,
  decodeToken,
  signToken
}
```

#### 反向代理信任配置 (trustProxy.js)

审计 H-8 新增，是 `TRUST_PROXY` 的**唯一**解析入口（`config/index.js` 只调用它，`server.js` 只消费解析结果）。

```javascript
// src/config/trustProxy.js（节选）
const NAMED_VALUES = ['loopback', 'linklocal', 'unixsocket'];
const OFF_VALUES = ['false', '0', 'off', 'none', 'no'];
const ALL_VALUES = ['true', '*', 'all'];   // 全量信任：一律拒绝
const MAX_HOPS = 10;

// 返回 { value, source }，source 为 'env' | 'default'，供启动日志与告警区分二者
function resolveTrustProxy(raw) {
  // 未设置 → { value: 'loopback', source: 'default' }
  // 'false'/'0'/'off'/'none'/'no' → { value: false, source: 'env' }
  // 'true'/'*'/'all' → throw（任何直连客户端都能伪造 req.ip 并绕过全部按 IP 计的限流）
  // 单个整数 1~10 → 跳数；0 → false；>10 → throw
  // 'loopback'、IP 字面量、CIDR（逗号分隔）→ 字符串或数组
}
```

取值规则与拒绝理由：

| 写法 | 结果 | 说明 |
| - | - | - |
| 未设置 | `loopback`（来源 default，启动告警） | 与仓库自带 nginx「同机 127.0.0.1 upstream」的拓扑一致，默认即正确 |
| `1` | 信任一跳 | 等价于改动前的写死值，但**必须由运维显式写出** |
| `false` / `0` / `off` | 完全忽略 `X-Forwarded-For` | 后端直接暴露给客户端时用；经反代部署会让所有请求共用一个限流桶（启动告警） |
| `loopback,172.16.0.0/12` | 信任回环 + 该网段 | docker-compose 编排采用（nginx 是另一个容器），见《部署文档》v6.10 |
| `true` / `*` / `all` / `0.0.0.0/0` | **拒绝启动** | 全量信任等于把 `req.ip` 交给请求方填写 |
| `123.45.67.89/24`（主机位非零） | **拒绝启动** | proxy-addr 会静默掩码成整段 `/24`，信任面被悄悄放大；提示改写为 `/32` 或网络地址 |
| `11`、`1,2`、`1,10.0.0.0/8`、八位组>255、前缀>32/128、乱串 | **拒绝启动** | 配置层报错远好过运行期限流静默失效 |

`config/index.js` 的相关导出：`trustProxy`（解析后的值）、`trustProxySource`（`'env'`/`'default'`）、
`nodeEnvExplicit`（`NODE_ENV` 是否由运维显式设置），以及 `rateLimit.disabled`（`RATE_LIMIT_DISABLED`）
与 `rateLimit.smsMax`（`SMS_RATE_LIMIT_MAX`）。

#### 认证中间件 (auth.js)

```javascript
// src/middlewares/auth.js
const { verifyToken, TOKEN_TYPES } = require('../config/jwt')
const { User } = require('../models')
const tokenRevocation = require('../services/tokenRevocation')

const authMiddleware = async (req, res, next) => {
  try {
    const token = req.headers.authorization?.replace('Bearer ', '')

    if (!token) {
      return res.status(401).json({ code: 401, message: '未提供认证令牌' })
    }

    // 审计 H-4：只接受 access 用途；refresh / admin 令牌在此返回 null，与验签失败同一分支
    const decoded = verifyToken(token, [TOKEN_TYPES.ACCESS])
    if (!decoded) {
      return res.status(401).json({ code: 401, message: '令牌无效或已过期' })
    }

    // 审计 H-5：吊销校验统一收进 tokenRevocation（单令牌黑名单 + 主体水位线），
    // 三种结果三种响应：已吊销 → 401；无法确认且生产（fail-closed）→ 503；无法确认且非生产 → 放行
    const revocation = await tokenRevocation.checkRevocation(token, decoded)
    if (!revocation.verifiable && tokenRevocation.revocationFailMode() === 'closed') {
      return res.status(503).json({ code: 503, message: '认证服务暂时不可用，请稍后重试' })
    }
    if (revocation.revoked) {
      return res.status(401).json({ code: 401, message: '令牌已失效' })
    }

    const user = await User.findByPk(decoded.userId)
    if (!user) {
      return res.status(401).json({ code: 401, message: '用户不存在' })
    }

    // status: 1=正常，其余=禁用
    if (user.status !== 1) {
      return res.status(403).json({ code: 403, message: '账号已被禁用' })
    }

    // jinyan_time: 禁言截止时间戳
    const now = Math.floor(Date.now() / 1000)
    if (user.jinyan_time && user.jinyan_time > now) {
      return res.status(403).json({ code: 403, message: '用户已被禁言' })
    }

    req.user = user
    req.userId = decoded.userId
    req.token = token
    next()
  } catch (error) {
    logger.error('认证中间件错误:', error)
    return res.status(500).json({ code: 500, message: '服务器内部错误' })
  }
}

// 可选认证：无 token / 令牌无效 / 已吊销 / 吊销状态无法确认时一律按「未登录」放行（不回 503，
// 它守的是公开页面），只在确认未吊销时才挂 req.user/req.userId。用途同样只认 [TOKEN_TYPES.ACCESS]。
const optionalAuth = async (req, res, next) => { /* 同上，省略 */ }

// 管理员认证（H-5 起为 async，因为每个候选令牌都要查一次吊销状态；M-2 起再回读一次数据库）：
// 支持 Authorization: Bearer 或 x-admin-token 两种传递方式
// 1. ADMIN_TOKEN 环境变量应急令牌（config.admin.token）→ req.adminTokenKind='emergency'
// 2. 管理员登录签发的 JWT：verifyToken(token, [TOKEN_TYPES.ADMIN]) 通过，
//    且 decoded.role === 'admin' 或 decoded.role_id === 1
//    （H-4：用户端 access/refresh 与管理员 admin_refresh 都进不来）
//    通过后先查吊销：已吊销 → 记 sawRevoked 并尝试下一个候选；无法确认且生产 → 503
// 3. 再按 decoded.adminScope 回读实时主体（M-2，见 resolveLiveAdmin 注释）：
//    account → Admin.findByPk + 角色权限 resolvePermissions 现场重算
//    role → AdminRole.findByPk 同口径；emergency → 仅当 config.admin.emergencyLogin 仍为 true
//    删号/旧令牌缺 adminScope → 401「令牌已失效」；被禁用 → 403「账号已被禁用」
//    回读失败：生产 fail-closed 503 / 非生产按令牌快照放行（与吊销检查同一取向）
// 4. 命中后把**实时主体**挂 req.admin，原文留在 req.adminToken / req.adminTokenKind='jwt'，供 /logout 使用
// 5. 响应区分：明确被吊销/主体已删除/旧格式 → 401「令牌已失效」；禁用 → 403「账号已被禁用」；用途不符/格式非法 → 403「无管理员权限」
```

```javascript
// src/services/chatService.js
const { ChatLog, ChatRoom, User, VirtualUser, ChatSession } = require('../models')

// 消息历史：ChatLog 列 fromid/toid/content/type/vod_url/sec/time/is_del/is_revoked

        { fromid: targetUserId, toid: userId }

    },
    offset,
    limit,
```

#### 通话计费服务 (callBillingService.js)

```javascript
// src/services/callBillingService.js
class CallBillingService {
  /**
   * 创建通话记录
   * @param {Object} params - 通话参数
   * @param {number} params.callerId - 主叫用户ID
   * @param {number} params.calleeId - 被叫用户ID
   * @param {number} params.type - 通话类型 1-语音 2-视频
   * @param {number} params.pricePerMin - 每分钟价格（金币）
   */
  async createCallRecord({ callerId, calleeId, type, pricePerMin }) {
    return await CallRecord.create({
      caller_id: callerId,
      callee_id: calleeId,
      type,
      price_per_min: pricePerMin,
      start_time: Date.now(),
      status: 1
    })
  }

  /**
   * 结束通话并结算费用
   * @param {number} recordId - 通话记录ID
   * @param {number} duration - 通话时长（秒）
   */
  async endCall(recordId, duration) {
    const record = await CallRecord.findByPk(recordId)
    const minutes = Math.ceil(duration / 60)
    const totalCost = minutes * record.price_per_min

    // 扣除主叫方余额
    await User.decrement('money', {
      by: totalCost,
      where: { id: record.caller_id }
    })

    // 增加被叫方收入
    await User.increment('money', {
      by: totalCost * 0.8, // 平台抽成20%
      where: { id: record.callee_id }
    })

    // 更新通话记录状态
    await record.update({
      duration,
      total_cost: totalCost,
      end_time: Date.now(),
      status: 2
    })

    // 记录计费明细
    await CallBilling.create({
      record_id: recordId,
      duration,
      unit_price: record.price_per_min,
      total_amount: totalCost,
      caller_id: record.caller_id,
      callee_id: record.callee_id
    })

    return totalCost
  }
}

module.exports = new CallBillingService()
```

> （`startCall` / `cancelCall` / `rejectCall` / `acceptCall` / `endCall` / `getCallHistory`），
> 既没有 `createCallRecord`，也没有 `price_per_min` 字段（计费口径在 `createBilling` 内以常量实现：
> 陪伴通话免费 180 秒、之后按分钟向上取整、单价 1 金币、陪玩分账 0.7，且 `totalFee<=0` 不建单）。

#### TRTC 房间成员校验（安全扫描 C-4，v6.8）

`trtcService.js` 原先既不查库也不比对身份：`GET /auth` 对任何登录用户无条件发放 UserSig，
`enterRoom(userId, roomId)` 只是把客户端传来的房间号写进返回对象，
`getRoomInfo(roomId)` 连 `userId` 都不收并把 `status` 写死为 `'active'`。
而「进哪个房间」由前端 URL 参数（`?room=`）决定 —— 等于通话房间在服务端完全不设防。

项目里没有「房间表」，房间成员的唯一真相来源是 `xn_call_record`
（`trtc_room_id` + `caller_id` + `callee_id`，由 `startCall` 写入）。因此新增两个闸门函数：

```javascript
// src/services/trtcService.js（节选）
const { CallRecord } = require('../models');
const ENTERABLE_CALL_STATUS = [0, 1];   // 0=呼叫中（主叫要先进房等待） 1=已接通

const resolveRoomCall = async (userId, roomId) => {
  const room = Number(roomId);
  const uid = Number(userId);
  if (!Number.isFinite(room) || room <= 0) throw new Error('房间号格式不正确');

  // 历史数据房间号可能重复（旧 startCall 用 Math.random 且不查重），按 status 升序取单：
  // 0/1 排在 2/3/4/5 之前，优先命中仍在进行的通话，不会被残留旧记录挡在门外。
  const call = await CallRecord.findOne({
    where: { trtc_room_id: room },
    order: [['status', 'ASC'], ['create_time', 'DESC']]
  });
  if (!call) throw new Error('房间不存在或已过期');
  if (Number(call.caller_id) !== uid && Number(call.callee_id) !== uid) throw new Error('无权进入该房间');
  return call;
};

const requireEnterableRoom = async (userId, roomId) => {
  const call = await resolveRoomCall(userId, roomId);
  if (!ENTERABLE_CALL_STATUS.includes(Number(call.status))) {
    throw new Error('通话已结束，无法进入该房间');
  }
  return call;
};
```

接入点：

| 位置 | 改动 |
|------|------|
| `controllers/trtc.js: getAuth` | 不传 `roomId` 只回「通道是否开通」（`userSig:null` + `requiresRoomId:true`）；传 `roomId` 先 `requireEnterableRoom` 再签票 |
| `trtcService.enterRoom` | `requireEnterableRoom` 后才生成 `userSig`/`roomSig`，并回带 `callId`/`callStatus` |
| `trtcService.leaveRoom` | 只 `resolveRoomCall`（挂断与离房顺序可能交错，结束后仍要能退房） |
| `trtcService.getRoomInfo` | 改签名 `(userId, roomId)`，返回 `CALL_STATUS_TEXT` 映射的真实状态与双方 ID |
| `trtcService.startBilling` | `requireEnterableRoom` 闸门 |
| `controllers/trtc.js: roomAccessErrorResponse` | 业务原因分流 403/404/422；四类之外的异常保持 500，不伪装成业务拒绝 |
| `callBillingService.allocateTrtcRoomId` | `crypto.randomInt(1, 1000000)` + 对 `status∈{0,1}` 通话查重、最多 5 次重试，冲突用尽则报「通话房间号分配失败」 |

**遗留**：①进房票据仍是房间无关的普通 UserSig，没有改用腾讯 PrivateMapKey 绑定房间——
现有 `generateRoomSig` 是官方**字符串房间号**的 userbuf 字节布局，而前端 `trtc.enterRoom({ roomId: Number(roomId) })` 进的是数字房间，
数字布局无法离线验证（需真实密钥 + SDK 联调），贸然切换会打断所有真实通话；
②`createRoom` / `startBilling` / `endBilling` 仍是「返回随机对象、从不落库」的占位实现，前端无调用方；
③socket 的 `call_invite` / `call_accept` / `webrtc_signal` 转发仍信任客户端上报的 `toId`（记入后续 H 项）。

#### 支付入账收口（安全扫描 H-2，v6.9）

`xn_order_chong` 的入账有两条入口：微信服务端通知 `POST /api/pay/wx-notify`（免登录，靠 MD5 签名识别来源）与
客户端回执 `POST /api/pay/wx-callback`（登录 + 反查）。v6.9 之前两条路径各写一遍「读状态→改状态→加币」，
于是同一个缺陷被复制了两次，且第二个缺陷只在客户端路径上：

```js
// 反例（v6.8 及以前，两条路径同构）：check-then-act
const order = await OrderChong.findOne({ where: { order_no: payNo, status: 0 } });
if (!order) throw new Error('订单不存在或已处理');
const transaction = await User.sequelize.transaction();
await order.update({ status: 1, pay_time: getTimestamp() }, { transaction }); // ← 无条件覆盖
await User.increment('money', { by: order.coins, where: { id: order.user_id }, transaction });
await transaction.commit();
```

`findOne(status:0)` 与 `update` 之间没有任何互斥：微信对同一笔订单会并发、多次重发通知，
两笔请求都能在对方提交前读到 `status=0`，于是**同一单加两次币**；
`update` 又不带 `status=0` 条件，写入的是内存里那份旧快照。
客户端路径还额外缺两道闸：路由不要求登录，订单也不校验属于谁——
订单号是 `DK` + 毫秒时间戳 + 4 字节随机，可枚举性虽低，但「任何拿到订单号的人都能替这单确认支付」本身就不该成立。

修复后与卡密核销 `consumeCardOnce` 用同一套原子占用写法（`services/payService.js`）：

```js
// settleRechargeOrder(order, verification)：verification 由服务端校验路径构造，服务层再兜一道
if (verification.verified !== true) throw payError(PAY_ERRORS.UNVERIFIED, '支付结果未经服务端校验，拒绝入账');
if (Number(verification.paidCents) !== Math.round(Number(order.amount) * 100)) {
  throw payError(PAY_ERRORS.AMOUNT_MISMATCH, '支付金额与订单不一致，已拒绝入账'); // 只比对，金额永不参与算币
}
return User.sequelize.transaction(async (transaction) => {
  const [affected] = await OrderChong.update(
    { status: 1, pay_time: now, update_time: now },
    { where: { id: order.id, status: 0 }, transaction }   // ← 抢单：只有从 0 改成 1 的那笔能继续
  );
  if (!affected) throw payError(PAY_ERRORS.SETTLED, '订单已入账');
  await User.increment('money', { by: Number(order.coins) || 0, where: { id: order.user_id }, transaction });
  return { orderNo: order.order_no, coins: Number(order.coins) || 0, duplicated: false };
});
```

InnoDB 下第二条 `UPDATE ... WHERE status=0` 会等第一笔事务的行锁释放后再重新判定条件，因此 `affected=0`，
抢输的一方走 `PAY_ORDER_ALREADY_SETTLED`；对 `/wx-callback` 这条 HTTP 路径按「幂等成功」（200 + `data.duplicated=true`）返回，
对 `/wx-notify` 则回 `SUCCESS` XML——否则微信会按退避策略无限重发同一通知。

**接入点**

| 位置 | 收口后的职责 |
|------|--------------|
| `services/payService.js: settleRechargeOrder` | 唯一的入账事务：校验来源 → 金额比对 → 原子抢单 → 加币 |
| `services/payService.js: wxPayCallback(userId, payNo, verification)` | 客户端回执：先查订单、再比 `user_id`，才进入账 |
| `services/payService.js: requireOwnedOrder(userId, orderNo)` | `/wx-query`、`/wx-close` 的归属闸门（在调支付平台之前） |
| `services/payService.js: getOrderStatus(userId, orderNo)` | 查询条件含 `user_id`，他人订单统一 404「订单不存在」 |
| `services/wechatPayService.js: handleNotify` | 只做验签与平台字段校验，入账一律委托给 `settleRechargeOrder` |
| `services/wechatPayService.js: queryOrder` | 回传 `paidCents`/`transactionId`，供 controller 构造 `verification` |
| `controllers/pay.js: payErrorResponse` | 按 `PAY_ERRORS` code 分流 403/404/422/200(幂等)，无 code 的意外故障保持 500 |
| `controllers/pay.js: wxNotify` | 未鉴权入口自身设防：请求体 64KB 上限、`'end'` 回调异常兜住、回显前拆 `]]>` |
| `routes/pay.js` | `/wx-callback` 补 `authMiddleware` |

**两处让「支付根本不可用」的缺陷（顺带修掉）**：
①`parseXml` 的正则 `/<(\w+)>([^<]+)<\/\1>/` 匹配不了微信真实报文的 `<tag><![CDATA[值]]></tag>`（CDATA 里有 `<`），
真实回调解析成空对象、`return_code` 恒为 `undefined`，永远在第一步被判失败——现在先剥 CDATA 包裹；
②`handleNotify` 只看 `return_code` 与签名，不看 `result_code`（真正的业务结果），
未付款/余额不足这类通知也会走到入账分支；现在 `return_code → sign → result_code → appid/mch_id → fee_type=CNY → total_fee 金额比对` 逐层判失败。

**遗留**：①`xn_order_chong` 没有支付流水号列，`transaction_id` 只进日志，对账需人工查日志（补列是一次增量 DDL）；
②`config.alipay` 三项无代码使用、也没有支付宝下单/回调路由，因此除 `ALLOW_UNVERIFIED_PAY_CALLBACK` 外支付宝渠道无法入账；
③微信 V2 的 MD5 签名待迁移 V3（需平台证书与 AES-256-GCM 解密回调报文），本轮只补 V2 报文内部的校验强度；
④`/wx-callback` 的入账仍只保证「同一订单不重复加币」，跨订单的平台流水号重放无法检测（同 ①，缺流水列）。

#### 后台登录防爆破（安全扫描 H-3，v6.10）

**问题**：`/api/admin/login` 与 `/api/admin-manage/login` 是同一份后台口令的两个入口，此前只被全局 `apiLimiter` 覆盖
（`src/config/index.js` 的 `rateLimit` 段默认是 100 次/60 秒，`.env` 配的是 1000 次/15 分钟，且当时的 `skipInDev` 在本地完全豁免——
该"按环境放行"的判据正是 H-8 的第二条根因，已于 v6.15 整体取消），
既没有登录专用限流，也没有失败次数概念——脚本可以不间断地在线爆破管理员口令。

**两道叠加的防护**

| 层 | 位置 | 口径 | 为什么要这一层 |
|----|------|------|----------------|
| 单 IP | `middlewares/rateLimit.js: adminLoginLimiter` | 15 分钟 5 次（`ADMIN_LOGIN_RATE_LIMIT_MAX`），超了 429；当时刻意是六个里唯一不带环境豁免的一个，v6.15 起六个统一没有豁免（见 H-8 章） | 挡掉单台机器的粗放爆破；`req.ip` 的可信度由 `TRUST_PROXY` 决定，见 2.1 章 |
| 单账号 | `services/adminLoginGuard.js` | Redis `admin:login:fail:<归一化账号>`，INCR + 首次失败补 15 分钟 TTL，累计 `ADMIN_LOGIN_MAX_FAILS`（默认 5）次即锁；成功登录 `DEL` | IP 限流换代理即可绕过，账号维度才锁得住 |

```javascript
// src/services/adminLoginGuard.js —— 控制器只用这三个函数
getLoginLockState(username)  // → { verifiable, locked, fails, retryAfterSec }
recordLoginFailure(username) // → 累计次数；Redis 不可用返回 null，不影响主流程
clearLoginFailures(username) // → 登录成功后清零
```

**四个刻意的设计决定**
1. **账号名归一化**：`String(username).trim().toLowerCase()`。`Admin` 与 `ADMIN` 若各记一份计数，攻击者只是多开一个计数器。
2. **不存在的用户名同样计数**（走同一条 401，文案不区分账号是否存在）。否则「这个账号会被锁」这件事本身就成了免费的账号存在性探测。
3. **缺参数的 400 不计失败**。如果空请求也算一次失败，任何人对 `admin` 发五次 `{}` 就能把真管理员锁在门外——防爆破的机制反过来变成 DoS。
4. **Redis 不可用 → 503 保守拒绝**（fail-closed）：`getRedisClient()` 为空、命令报错或读写超过 2 秒都归为 `verifiable:false`，
   控制器记录 `logger.error` 并返回 503「登录校验服务暂不可用」，不做 fail-open。这与 `smsService` 的验证码校验同一口径——
   安全控制不能因为缓存抖动静默失效。`withRedisTimeout` 用 `{ value }` 包一层，把「键不存在（值为 null）」和「命令超时/报错」区分开。

#### 令牌用途绑定（安全扫描 H-4，v6.11）

**根因**：四类令牌（用户 access、用户 refresh、管理员 access、管理员 refresh）都由同一个 `config.jwt.secret` 签名，
payload 里**没有任何用途标记**，`verifyToken()` 也只看验签结果。后果是：

- refresh token 默认 30 天有效，却可以直接当 `Authorization: Bearer` 使用——一旦泄漏，可用期从 7 天放大到 30 天；
- 管理员令牌与用户令牌同构，`authMiddleware` 不区分用途；`adminAuth` 只看 payload 里的 `role`/`role_id`，
  而这两个字段本身没有任何「这是管理员令牌」的密码学约束；
- 任何一处漏带校验（例如新增入口直接 `jwt.verify`）都会把所有令牌变成通用凭证。

**做法**：签发端强制写 `type`，校验端强制传用途白名单；用途不符按验签失败同一处理（401 / 403 / `UNAUTHORIZED`），不做兼容降级。

| 文件 | 改动 |
|------|------|
| `src/config/jwt.js` | 新增并导出 `TOKEN_TYPES`；`generateToken`/`signToken` 经 `assertTokenType` 强制合法 type（未知或缺省直接抛错）；`verifyToken(token, allowedTypes)` 支持用途白名单，省略第二参时只验签（保持既有调用方语义）；`refreshAccessToken` 只接受 `type=refresh` |
| `src/middlewares/auth.js` | `authMiddleware`、`optionalAuth` → `[TOKEN_TYPES.ACCESS]`；`adminAuth` → `[TOKEN_TYPES.ADMIN]`，并去掉函数内重复的 `require('../config/jwt')` |
| `src/socket/index.js` | 握手鉴权 → `[TOKEN_TYPES.ACCESS]` |
| `src/services/authService.js` | 刷新后复校新 access 令牌 → `[TOKEN_TYPES.ACCESS]`；移除未使用的 `generateToken` 引入 |
| `src/controllers/admin.js`（3 处）、`src/controllers/adminManage.js`（5 处） | `signToken(..., TOKEN_TYPES.ADMIN / ADMIN_REFRESH)` |
| `src/controllers/user.js` | 移除从未使用的 `generateToken` 引入 |

**设计决定**

1. **白名单作第二参而非另起 API**：`verifyTokenFor(type, token)` 这类平行函数会让「忘记传用途」成为静默失败；
   默认 `null` 只影响显式省略，且鉴权入口改完后 grep 一轮即可确认无遗漏。
2. **签发端抛错而非回落默认值**：`signToken` 历史上被管理端各处传不同 `expiresIn` 调用，若 type 缺省回落成 `access`，
   就会签出「名义 access、实际给管理端用」的令牌，正好复刻本次要修的缺陷。
3. **重签只结转 `userId`**：旧代码 `const { userId, exp, iat, ...payload } = decoded` 里的 `payload` 是从未使用的死变量，
   后续改动若直接拿它签名，refresh 令牌里伪造的 `role`/`permissions` 就会被洗进新的 access 令牌。
4. **不兼容历史令牌**：加「`type` 缺失视为 access」的兼容层等于保留攻击面。上线后所有既有会话需重新登录，
   属预期行为变更（前端 `common.js` 的 401 分支已清登录态并跳转登录页）。

**遗留**：①`admin_refresh` 令牌目前**没有服务端消费入口**（`/api/admin-manage/login` 返回它，管理端只写进
`localStorage`，无任何接口校验它）——v6.12 的 H-5 已让登出时把它一并作废（需 `type=admin_refresh` 且 `id` 归属本人），
但**轮换与日常校验仍无**；②`JWT_EXPIRES_IN` 默认 `7d`，access 令牌时效偏长，属运维口径（v6.12 起服务端已具备即时吊销手段，缩短默认值不再有「无法踢人」的顾虑）；
③除登出黑名单外仍无主动吊销手段——**已由 v6.12（H-5）收口**，见下一小节；④新增 16 个 jest 用例（`config/jwt` +8、`middlewares/auth` +6、
`services/authSmsCode` +2）并强化 6 个既有用例（`controllers/admin` 3 例、`controllers/adminManage` 3 例断言签发 type），
因本机无 node/npm **未执行**，仅静态复核。

#### 服务端登出与令牌吊销（安全扫描 H-5，v6.12）

**根因**：`authMiddleware` 里那段 Redis 黑名单读取**全仓库没有任何写入点**，也没有登出接口——
「退出登录」只是前端清 `localStorage`，令牌在剩余有效期内（access 默认 `7d`）依然可以直接打接口。
连带三个后果：①被盗令牌不会因为用户登出而失效；②改密/重置密码/封号/删除管理员都踢不掉已签发的令牌
（`adminAuth` 只验 JWT、**不回查数据库**，权限还是登录时的快照）；③Socket 握手同样不看黑名单，
登出后的令牌还能继续收私聊与通话信令。

**做法**：新增 `src/services/tokenRevocation.js` 作为唯一的吊销读写入口，写入侧接登出与四类会话终止动作，
读取侧接三个鉴权中间件 + 刷新链路 + Socket 握手。

| 文件 | 改动 |
|------|------|
| `src/services/tokenRevocation.js`（新增） | 两类吊销状态 + 统一读写；`withRedisTimeout` 用 `{ value }` 包裹以区分「键不存在（可确认未吊销）」与「命令报错/超 2 秒（不可确认）」 |
| `src/middlewares/auth.js` | `authMiddleware`/`optionalAuth` 改调 `checkRevocation`；`adminAuth` 转 **async**，逐个候选令牌查吊销，并留下 `req.adminToken`/`req.adminTokenKind` |
| `src/controllers/user.js` | 新增 `logout`；`resetPassword` 改密**前**先做全端下线；`refreshToken` 错误映射补 `REFRESH_REVOKED`→401、`REVOCATION_UNAVAILABLE`→503 |
| `src/services/authService.js` | `refreshToken` 先按 `[REFRESH]` 复校用途，再查吊销，最后才重签 |
| `src/controllers/adminManage.js` | 新增 `adminLogout`；`updateAdminPassword`/`deleteAdmin`/`updateAdmin(status≠1)` 三处改库前先吊销该账号全部会话 |

| `src/socket/index.js` | 握手增加吊销校验，已登出 → `UNAUTHORIZED`，无法确认且生产 → `AUTH_SERVICE_UNAVAILABLE` |
| `src/services/index.js`（附带修复） | `levelService` 被 `require` 却从未 `export`，导致 `controllers/{user,circle,album}.js` 里解构出的 `levelService` 是 `undefined`，所有等级门槛接口必然 TypeError→500 |

**两类吊销状态（都只存 Redis）**

| 键 | 值与 TTL | 语义 |
|----|----------|------|
| `auth:blacklist:<sha256(令牌原文)>` | `'1'`；TTL = `exp - now`，上限 `refreshExpiresIn` | 单令牌拉黑（登出当前设备），令牌自然过期即自动清理 |
| `auth:revoke-user:<userId>` / `auth:revoke-admin:<adminId>` | 水位线秒数（取 `now + 1`）；TTL = `refreshExpiresIn` | 「全端下线」：`iat` 早于水位线的令牌一律作废 |

**设计决定**

1. **键里放摘要不放原文**：原始 JWT 是可直接复用的凭据，放进键名会出现在 `MONITOR` 输出、`KEYS` 结果和 RDB 备份里，
   等于把凭据复制进一个通常不设防的存储。
2. **水位线 TTL 取 `refreshExpiresIn` 而非 `expiresIn`**：全端下线后刷新链路必须仍然被拦住，
   而水位线要活过「被它作废的那批令牌里最长寿的一枚」；`expiresIn`/`refreshExpiresIn` 是 `7d`/`30d` 这类字符串，
   所以 `parseExpiresSeconds` 必须真解析，解析不出来才回退默认值。
3. **`iat` 水位线取 `now + 1` 秒**：`iat` 只有秒级精度，取 `now` 会让同一秒内刚签发的令牌逃过吊销。
4. **只登记验签通过的令牌**：否则任何人送一枚伪造串就能凭超大 `exp` 往 Redis 里塞一个长期键。
5. **登出只吊销本人凭据**：请求体里带的 `refreshToken` 必须先 `verifyToken(..., [REFRESH])` 且
   `userId`/`id` 与当前登录者一致才拉黑——否则这个口子会变成「替别人登出」的定向 DoS。
6. **C 端登出走水位线（`logoutAllDevices`）**：登录响应下发了 `refreshToken`，但 C 端 SPA 从不保存它，
   「只拉黑 access」会留下一枚 30 天内还能换新令牌的孤儿凭据；代价是**用户端登出 = 该账号全端下线**，
   不是单设备语义。
7. **写库前吊销，且写入侧与读取侧同一 fail 取向**：生产 `fail-closed`（确认不了就不改密/不删号，回 503），
   非生产 `fail-open` 让流程走完并用 `devicesRevoked`/`sessionsRevoked`/`serverRevoked` 如实回报——
   本地与 CI 常常没有 Redis，写入侧若硬失败会把「重置密码」「改管理员口令」变成不可用功能。

**行为变更**：①新增 3 个登出接口；②`POST /api/user/reset-password` 成功后返回 `data.devicesRevoked`，
且该账号此前签发的全部令牌（含 refresh）作废；③改/重置口令、封号、禁用与删除管理员都会终止对应会话；
④生产环境下 **Redis 不可达 = 所有需登录接口 503**（此前是「Redis 客户端为空就整段跳过黑名单检查直接放行」，
等于打挂 Redis 可让已登出令牌全部复活）；⑤`adminAuth` 的明确吊销返回 401「令牌已失效」（用途不符仍是 403），
管理端 `common.js` 的 401 分支会清 `admin_token`/`admin_refresh_token`/`admin_user` 三件套并跳登录页。

**遗留**：①`ADMIN_TOKEN` 应急固定令牌**不可吊销**（环境变量里的固定字符串，无 `exp` 无签名），
用它登录时 `/logout` 返回 `revocable:false`，只能清前端本地态；②修改管理员**权限/角色**（`updateAdmin` 的
`permissions`、`role_id`）仍不吊销在用令牌——这是登录时快照的另一半，本轮故意不做，避免管理员改自己的权限时
把自己踢下线，需产品确认后另做；③`admin_refresh` 仍无服务端消费入口，本轮只补了吊销能力；
④Socket 握手的吊销分支**无单测**（该模块此前无任何测试），需人工联调；
⑤`mock Redis`（`USE_MOCK_DB=true`）的内存客户端没有 `isReady`，`getRedisClient()` 因此返回 `null`，
且缺 `incr`/`ttl`——mock 模式下吊销写入统一走 fail-open，短信验证码与后台登录防爆破同样不可用，需真实 Redis；
⑥本轮改写/新增的用例（按文件当前 `it()` 总数计，多轮改动未逐轮拆分）：`services/tokenRevocation.test.js`（本轮新建，23 例）、`middlewares/auth.test.js`（32 例，`adminAuth` 全部转 async）、
`controllers/user.test.js`（77 例，含登出与重置密码的 fail-open/closed 双路）、`controllers/adminManage.test.js`（53 例，mock 补 `revocationFailMode`，含改密/删号/禁用的双路）、
`controllers/admin.test.js`（69 例，含封号尽力而为下线）、`services/authSmsCode.test.js`（11 例，本轮重写其中 1 组 7 例）、

#### 装扮商城服务端化（安全扫描 H-6，v6.13）

**根因**：付费装扮的「价格」「扣款额」「所有权」三件事都由客户端提交——
`POST /api/user/pref` 的 `body.spend` 被直接当作扣款额（`controllers/user.js` 只判断余额够不够），
`body.data.skinShop.owned/using` 原样 `Object.assign` 进 `xn_user_pref.data`。
抓包把 `spend` 改成 `0`、`owned` 全填 `true`，就能一分钱不花拿到 8888/9999 币的装扮。
更糟的是等级门槛也挂在这条链上：`benefitsRequested()` 里的判定是 `data.skinShop && spendNum > 0`，
所以 `spend=0` 不仅跳过扣费，连 `skin_shop`（Lv.45）的 `requireLevel` 都一起绕过。
服务端从来没有装扮目录——价格只存在于 `frontend/src/views/SkinShop.vue` 的硬编码数组里。

**做法**：把目录、价格、VIP 门槛、所有权、扣款金额五件事全部收进服务端，客户端只提交 `skinId`。

| 文件 | 改动 |
|------|------|
| `src/utils/skinCatalog.js`（新增） | 26 项装扮的唯一事实来源（`frame` 8 / `badge` 6 / `bubble` 6 / `theme` 6），每项含 `id/type/name/price/vipLevel/assets` 与派生的 `key=<type>_<id>`、`vipOnly`；`findSkin`/`findSkinByKey` **只接受字符串**，防止带 `toString` 的对象被当成 id；`isFreeSkin` 定义「免费且非 VIP 专属」 |

| `frontend/src/services/skinService.js`（新增）、`views/SkinShop.vue`、`services/userPrefService.js`、`views/{AvatarFrame,IdentityBadge,PriorityMatching,StealthVisit}.vue` | 见《前端技术文档》v7.19 / 《前端功能拆解开发指导》v6.21 |

**设计决定**

1. **目录放代码而不是建表**：装扮是随版本发布的静态内容，改价应走代码评审与发布；
   运营在线调价需要新表 + 管理端 CRUD + 迁移，超出本轮范围（记入遗留）。
2. **所有权仍存 `xn_user_pref.data.skinShop`**：避免迁移与历史数据搬迁，代价是它和普通偏好同居一段 JSON——
   因此通用偏好接口必须**显式剥离**该键，写入权只留给 `skinService`。
3. **状态键沿用 `<type>_<id>`**：改造前前端就是这套键，历史数据才还能被 `sanitizeState` 读出来。
4. **免费装扮不写购买记录**：`isFreeSkin` 的项直接派生为「已拥有」，不必为默认外观播种；
   但领取仍走 `purchase`，所以**同样过 Lv.45 门槛**——这正是改造前被 `spend=0` 绕开的那条判定。
5. **切换已拥有装扮不查门槛，购买必查**：沿用改造前「关闭隐身、切换已有装扮不该被门槛挡住」的口径，
   门槛只约束「拿到新东西」。
6. **`spend` 只容忍显式 `0`**：旧客户端仍会带 `spend:0`，直接放行；`abc`/`-100`/`'500'` 一律 422 并指路到新接口，
   避免「静默忽略」让调用方以为自己扣款成功。
7. **用户行 `LOCK.UPDATE` + 条件扣减**：并发重复提交同一装扮时，第二个请求在锁后能读到已写入的 `owned`，
   走幂等分支不再扣款；余额不足用 `WHERE money >= price` 的 `decrement` 判定，扣款与所有权同生共死。

#### 预约/订单状态机幂等（安全扫描 H-7，v6.14）

**根因**：钱相关的状态迁移都是「先 `findByPk` 读状态 → 判一遍 → 无条件 `update` 改状态 → `User.increment` 动钱」。
两条语句之间没有互斥，所以同一笔单并发两次（双击、前端重试、买卖双方在各自端同时点「完成」）会**各自通过校验**，
于是同一笔预约退两次款、给陪玩师分两次账。同项目 `gamesService.cancelOrder`（审计 B-02）、`grabOrder`、
`payService.settleRechargeOrder`（H-2）、`walletService.applyWithdraw`（C-2）、`skinService.purchase`（H-6）
早已改用「条件更新 + 影响行数」判据，这几处属遗漏。

**做法**：所有动钱的状态迁移改为一句话抢占——`Model.update({新状态}, { where: { id, status: <期望前态>, 归属列 }, transaction })`，
`affectedRows === 0` 直接抛错回滚，命中那一次才继续动钱；读-改-写的聚合列（陪玩师 `income_total/order_num/star/pingjia_num`）
在同一事务里用 `SELECT ... FOR UPDATE` 锁行后重算。

| 位置 | 迁移 | 改后口径 |
|------|------|----------|
| `reserveService.cancelReserve` | `0\|1 → 4` + 按比例退款 | 条件更新带 `user_id` 归属；未命中 → 「预约无法取消」并回滚，退款不再重复 |
| `reserveService.completeReserve` | `1 → 3` + 分账 | 条件更新 `WHERE id AND status=1 AND (user_id=? OR target_user_id=?)`；`reserve_commission_rate` **移入事务内**读取 |
| `reserveService.confirmReserve` | `0 → 1` | 条件更新带 `target_user_id` |
| `reserveService.rejectReserve` | `0 → 2` + 释放时间槽 | 原来**完全无事务**，现「改单 + 释放槽位」同事务，且释放只命中 `status=1` 的占用槽 |
| `reserveService.toggleSlot` | `0 ⇄ 2` | **`status=1`（已成交）直接拒绝**——原实现会把 1 翻回 0，凭空释放已占用时段 → 同一时段被重复预约；并发下状态已变则提示刷新重试 |
| `gamesService.startOrder` | `1 → 2` | 条件更新带 `target_user_id` |
| `gamesService.completeOrder` | `2 → 3` + 分账 + 资料累加 | 条件更新带 `user_id`；`order_commission_rate` 移入事务内；陪玩师行 `LOCK.UPDATE` 后再算 `income_total/order_num`；经验 `+50` 仍在提交后，只有抢占成功方会走到 |
| `gamesService.appealOrder` | `1\|2\|3 → 5` | 条件更新，避免「已完成又被改回申诉中」的覆盖 |
| `gamesService.evaluateOrder` | 写评价 + 滚动平均分 | 事务内 `FOR UPDATE` 重读订单行与陪玩师资料行；重复评价返回「该订单已评价」，平均分不再被同一单计两次 |
| `settingsService.getCommissionRate` | — | 新增可选第三参 `transaction`，让「读比例」与「写金额」同快照；省略时行为不变 |

**设计决定**

1. **判据用影响行数，不用实例 `save()`**：Sequelize 的 `instance.update()` 无法表达 `WHERE status=?`，
   条件更新是唯一能把「前态」写进 SQL 的写法（`Model.update` 在 MySQL 下返回 `[affectedRows]`）。
2. **归属列一并进 `WHERE`**：越权校验（`user_id`/`target_user_id`）在读和写两处都保留，读侧负责给出准确文案，
   写侧负责并发下不被绕开——纵深而非重复。
3. **先抢占再动钱，钱必须在同一事务内**：抢占成功但打款失败要能整体回滚；反之若先打款再改状态，
   落败方仍会重复入账（这正是改造前的顺序）。
4. **失败复用既有错误文案**：接口路径、请求体、成功载荷与错误码分类不变，客户端无需发版；
   只有 `toggleSlot` 新增两条拒绝文案（它此前能把已成交时段改回可预约）。
5. **`getCommissionRate` 的事务参数可选**：其余调用方（提现、审核退回等）不需要同快照，省略即维持原语义。

**行为变更**：①`/api/reserve/slots/toggle` 对已成交时段返回「该时间段已被预约，不可修改」（该控制器业务错误走 **422**）；

#### 红包领取并发收口（安全扫描 M-7，v6.22）

**根因**：`giftService.receiveRedPacket` 把「读红包」和「算金额」放在了事务**之外**——
先 `RedPacket.findOne({ where: { packet_no } })` 拿到红包，用它的 `remain_amount` 为拼手气红包取随机金额，
**然后**才开事务，用「`remain_num > 0`」的条件更新去占用名额（并把 `remain_amount` 用 `GREATEST(remain_amount - amount, 0)` 夹到 ≥0）。
条件更新只保证「名额不会被抢成负数」，**保不住金额**：`GREATEST` 把剩余额夹到 0 时，多出来的那份仍然照常 `User.increment` 进钱包。
两个并发请求都拿着同一份旧 `remain_amount` 取随机数，就能合计发出多于 `total_amount` 的金额（**超额发放**）。
同项目 `skinService.purchase`（H-6）、`gamesService.completeOrder`（H-7）早已用「事务内锁行/条件更新」写资金路径，本处属遗漏。

**做法**：把整个领取过程收进**一个事务**，进事务后第一件事就是**锁定红包行**再算金额：

```javascript
// src/services/giftService.js（节选）
const transaction = await sequelize.transaction()
const lock = transaction.LOCK ? transaction.LOCK.UPDATE : undefined
const packet = await RedPacket.findOne({ where: { packet_no: packetNo }, transaction, lock }) // SELECT ... FOR UPDATE
// ...状态/有效期/不能领自己/重复领取校验（均在事务内）
const remainNum = Number(packet.remain_num)
const remainCents = Math.floor(Number(packet.remain_amount) * 100 + 1e-6)   // 以「分」计算，消浮点误差
let amountCents
if (remainNum === 1) amountCents = remainCents                              // 最后一人拿走剩余全部
else if (packet.type === 1) amountCents = crypto.randomInt(1, Math.max(1, remainCents - (remainNum - 1)) + 1) // 拼手气：为其余人各留 ≥1 分
else amountCents = Math.floor(remainCents / remainNum)                      // 普通：均分，不足留给最后一人
const amount = Math.min(amountCents, remainCents) / 100
```

**设计决定**

1. **用 `SELECT ... FOR UPDATE` 而不是只靠条件更新**：并发同一红包的领取在行锁上排队，后到者读到的是已被扣减的
   `remain_num`/`remain_amount`，分配天然基于最新余额——这是「金额也正确」的前提；条件更新（`remain_num > 0`）保留为第二道闸。
2. **一律以「分」为单位**：金额随机/均分都在整数分上做，最后除以 100，避免 `0.1 + 0.2` 这类浮点误差累积。
3. **最后一人必须取剩余全部**：收尾的「名额领完则关闭红包」会把 `remain_amount` 置 0，若不把零头发给最后一人，
   这部分钱会凭空消失（旧实现正是如此——普通红包 `floor(total/num)` 的零头永远发不出去）。
4. **重复领取靠唯一索引兜底**：业务判定给出友好文案，`(packet_id, user_id)` 唯一索引在并发下兜住漏网，
   `ER_DUP_ENTRY` 统一转为「已领取过该红包」。
5. **顺带对齐字段名并挂校验**：`/api/gift/redpacket/receive` 现由 `validate` 要求 `packetNo`（必填字符串），
   前端 `giftService.receiveRedPacket` 同步由 `packetId` 改名为 `packetNo`（M-6 登记的历史缺陷）。

**行为变更**：①并发抢同一拼手气红包不再超额发放（原先可刷出多于 `total_amount` 的金额）；
②普通红包零头改由最后一名领取者获得（原实现零头被清零、少发）；③`packetNo` 缺失/非字符串由 422 变 400。
接口路径、成功载荷（`data.amount`）、其余错误码分类不变，客户端无需发版。

**遗留**：①C 端「领取红包」入口仍未接线——`frontend/src/views/ChatRoom.vue` 的红包气泡没有点击领取的处理函数，
`frontend/src/services/giftService.js` 的 `receiveRedPacket` 在全仓库**无任何调用方**，本接口此前只能被直接调用
（本次只收口服务端并发与契约，**未新增前端领取交互**，属独立前端待办）；②红包记录 `GET /api/gift/redpacket/history` 未分页；
③`sendRedPacket` 的 `totalAmount`/`totalNum` 由发送方提交、服务端只做量级校验（红包无「定价」，属正常）。
本轮改写 `tests/unit/services/giftService.test.js` 的 `receiveRedPacket` describe（原 5 例 → 现 12 例）因本机无 node/npm **未执行**，仅静态复核。

#### 反向代理信任与限流豁免（安全扫描 H-8，v6.15）

**根因**（两条独立缺陷叠在一起，任一条都能让限流归零）：

1. `server.js` 写死 `app.set('trust proxy', 1)`。Express 一旦信任跳数，`req.ip` 就取自客户端可任意填写的
   `X-Forwarded-For`；只要 origin 本身可达（容器端口映射到公网、内网横向、nginx 之外的第二个入口），
   加一跳伪造地址即可每个请求换一个 `req.ip`，**六个按 IP 计的限流全部失效**，
   访问日志（`server.js:225`）与 `xn_admin.last_login_ip`（`controllers/adminManage.js:117`）也随之可被涂抹。
2. `middlewares/rateLimit.js` 六个限流器有五个挂 `skip: (req) => config.nodeEnv === 'development'`，
   而 `config/index.js` 里 `nodeEnv` 的兜底值就是 `'development'` —— 忘设 `NODE_ENV` 的生产部署
   会在自己的生产机器上把全局/登录/短信/卡密/上传五道限流整体放行，且没有任何日志提示。

**做法**：信任层级改为「必须显式声明 + 默认收敛到最小」，环境豁免改为「唯一显式开关 + 生产硬拦」。

| 文件 | 改动 |
|------|------|
| `src/config/trustProxy.js`（新增） | `TRUST_PROXY` 唯一解析入口，默认 `loopback` 并回报来源；`true`/星号/`all`/`0.0.0.0/0`、跳数>10、跳数与网段混写、八位组>255、前缀越界、**主机位非零的 IPv4 网段**全部抛错拒绝启动（详见 2.1 章取值表） |
| `src/config/index.js` | 导出 `trustProxy`/`trustProxySource`/`nodeEnvExplicit`；`rateLimit` 段新增 `disabled`（`RATE_LIMIT_DISABLED`）与 `smsMax`（`SMS_RATE_LIMIT_MAX`） |
| `src/middlewares/rateLimit.js` | 删除 `skipInDev`，六个限流器统一 `skip = () => config.rateLimit.disabled === true`；统一 `standardHeaders:true`/`legacyHeaders:false`（此前只有 `apiLimiter`、`adminLoginLimiter` 两个配了标准头，其余四个走已废弃的 `X-RateLimit-*`）；`smsLimiter.max` 由硬编码 `1` 改读 `config.rateLimit.smsMax` |
| `server.js` | `app.set('trust proxy', config.trustProxy)` + 启动打印取值与来源；`enforceProductionConfig()` 新增致命项（生产 `RATE_LIMIT_DISABLED=true` 拒绝启动）；`collectConfigRisks()` 新增 4 条告警（`NODE_ENV` 未显式设置 / `TRUST_PROXY` 仍是默认值 / `TRUST_PROXY=false` / 限流已关闭） |
| `docker-compose.yml` | backend 注入 `TRUST_PROXY=${TRUST_PROXY:-loopback,172.16.0.0/12}`（该编排里 nginx 是另一个容器，只信任回环会让全站共用一个限流桶） |
| 根与 backend 的 `.env.example` / `.env.production.example` | 补 `TRUST_PROXY`、`RATE_LIMIT_DISABLED`、`SMS_RATE_LIMIT_MAX`、`LOGIN_RATE_LIMIT_MAX` 键位与注释（只写键名与说明，不含任何真实凭据） |
| `tests/integration/functional-app.js` | 原先依赖 dev 自动豁免，现显式 `RATE_LIMIT_DISABLED=true`，且必须放在 `require('../../src/routes')` **之前**（`src/config` 在 require 期即读取环境变量） |

**设计决定**

1. **默认 `loopback` 而不是跳数 1**：仓库自带 nginx 的 upstream 走 `127.0.0.1`，只信任回环即可在
   `$proxy_add_x_forwarded_for` 追加的链尾拿到真实客户端 IP；同时任何直连 origin 者送上的 XFF 一律被忽略 —— 
   即「默认值既正确又 fail-closed」。
2. **危险取值在配置层抛错，不做静默降级**：`trust proxy=true` 会让 express-rate-limit 直接抛
   `ERR_ERL_PERMISSIVE_TRUST_PROXY`，而跳数/网段写错是**无声**的；启动失败比限流静默失效可查得多。
3. **主机位非零的网段一并拒绝**：`123.45.67.89/24` 经 proxy-addr 掩码后实际信任整段 `/24`，
   与运维写下的「就这一台」不符 —— 属于配置层面的信任面放大，要求写成 `/32` 或规范网络地址。
4. **取消环境豁免而不是"改成判断得更准"**：`NODE_ENV` 缺省即 `development` 是 Express 生态的既成事实，
   以它作为安全开关的语义前提太脆弱；换成显式 `RATE_LIMIT_DISABLED`，并让生产环境设它即拒绝启动。
5. **六个限流器同一套响应头**：`standardHeaders:true` 在 express-rate-limit 7.5.1 里解析为 `draft-6`，
   正常响应即带 `RateLimit-Policy`/`RateLimit-Limit`/`RateLimit-Remaining`/`RateLimit-Reset`，
   429 额外带 `Retry-After`（`legacyHeaders` 或 `standardHeaders` 任一开启即写入），
   客户端不必再猜恢复时间（此前 `loginLimiter`/`smsLimiter`/`cardLimiter`/`uploadLimiter` 四个只有旧的 `X-RateLimit-*`）。

**行为变更**：①本地 `NODE_ENV=development` 不再自动豁免任何限流（验证码 1 次/分钟、登录 20 次/15 分钟、
后台登录 5 次/15 分钟、卡密 10 次/分钟、上传 20 次/分钟在开发机同样生效），需要放开用
`SMS_RATE_LIMIT_MAX` / `LOGIN_RATE_LIMIT_MAX` / `ADMIN_LOGIN_RATE_LIMIT_MAX` / `RATE_LIMIT_MAX_REQUESTS`，
或本地整体 `RATE_LIMIT_DISABLED=true`；②启动日志新增一行 `🔒 trust proxy = <取值>（来源：…）`；
③生产环境 `RATE_LIMIT_DISABLED=true` 变为致命配置；④`TRUST_PROXY=true`/`*`/`all` 等写法由"能用"变为"拒绝启动"。
本轮**无库表变更、无迁移、无新接口**。

**遗留**：①限流计数仍是进程内存（`express-rate-limit` 默认 MemoryStore），多实例部署时每实例各一份，
要全站一致需接共享 Redis store，属独立改造项；②`LOGIN_RATE_LIMIT_MAX` / `ADMIN_LOGIN_RATE_LIMIT_MAX`
仍在中间件内直接读 `process.env`，未收敛进 `config`（与 `SMS_RATE_LIMIT_MAX` 口径不一致）；
③`X-Real-IP` 未被使用（Express 只认 XFF），nginx 两个头都发，无需改动；
④`tests/e2e/{api-matrix,real-data-smoke,real-data-write}.js` 面向开发服务器的压测式脚本，
现在会真吃到 429，运行前需以 `RATE_LIMIT_DISABLED=true` 启动后端（已写入《部署文档》v6.10 与《后端交互功能梳理》）；

#### .env 卫生（安全扫描 H-9，v6.16）

**扫描事实**

1. 工作区 `.env`（根与 `backend/`）存有真实可用凭据（云数据库 root、Redis/Mongo 口令、模型平台 API Key）。
   两个文件均被 `.gitignore` 忽略、**从未入库**（`git log --all -- .env` 为空），实质风险是「弱值 + 长期不轮换」，
   而非「已随代码泄露」。
2. `.env` 里 `NODE_ENV=development`，不会触发 `enforceProductionConfig()`（该函数只在 `production` 下执行），
   弱密钥只在启动日志告警。告警通道本身 H-8 轮已建：`warnWeakSecrets()` 在**任何环境**打印弱值清单，
   另有 `NODE_ENV` 未显式设置的独立提示；生产强校验（`config/index.js:7-16`：`JWT_SECRET`/`DB_PASSWORD`
   必填、`JWT_SECRET` ≥32 字符，缺一即拒绝启动）维持不变。
3. 四份 `.env*.example` 模板缺键（照模板部署会漏配）且含死键。

**做法**：删除「静默弱口令兜底」，补齐键位清单，无法用 compose 强校验的缺口以启动告警补位。

| 文件 | 改动 |
|------|------|
| `src/config/index.js` | `db.mysql.password` 删除兜底 `'123456'` → `process.env.DB_PASSWORD \|\| undefined`（未配置时连接直接失败，不再静默用弱口令连库） |
| `scripts/backup.js` | docker 模式 Redis 备份删除兜底 `'redis123'`：未设置 `REDIS_PASSWORD` 时打印错误并让该步显式失败返回 |
| `docker-compose.yml` | coturn 的 `${TURN_SECRET:-change-me-in-env}` 兜底**保留**（原因见设计决定 1），注释标注 H-9 防护去向 |
| `server.js` | `collectConfigRisks()` 新增 TURN 告警：仅当 `TURN_URL` 已配置而 `TURN_SECRET` 缺失/弱值时提示「中继凭据可被冒签」 |
| `scripts/deploy-selfcheck.js` | 弱密钥扫描补 `MONGO_URI` 内嵌口令特征检查（`:admin123@`/`:changeme@`/`:password@`/`:123456@`），与 `server.js` 同一组特征 |
| 根/backend × `.env.example`/`.env.production.example` | 补齐全量键位：`SERVER_URL`、`ADMIN_USERNAME`/`ADMIN_PASSWORD`、`WECHAT_NOTIFY_URL`/`WECHAT_CERT_PATH`、`ALIPAY_*`×3、`STORAGE_FALLBACK` 与 QINIU/OSS 键组、`CALL_CHANNEL`/`STUN_URLS`/`TURN_*`、`LLM_*`×7、`VIRTUAL_USER_CHAT_HISTORY_LIMIT`、`AUTO_MIGRATE`/`ENABLE_SWAGGER`/`SWAGGER_PROD_URL`/`STRICT_ROUTES`、`LOGS_PATH`/`LOG_TO_CONSOLE`、注释形式 `BACKUP_*`×4；删除死键 `APP_NAME`、`SMS_SDK_APP_ID`（后者为 C-1 轮误加，短信代码只读 `SMS_APP_ID`）；只写键名与注释，不含任何真实值 |
| `tests/unit/config/envHygiene.test.js`（新增） | 4 例锁定「配置层不再含口令兜底」 |

**设计决定**

1. **coturn 为什么保留兜底**：compose 的变量插值先于 profile 过滤执行 —— `${TURN_SECRET:?}` 会让未启用
   `turn` profile 的普通部署也在 `docker compose up` 解析阶段一并报错。防护改放后端：`TURN_URL` 已配置
   （= 中继真的启用）而密钥为弱值时启动告警；未部署 TURN 时该键为空属正常状态，不告警。
2. **删兜底而不是删键**：`password: ... || undefined` 让「忘配」变成可见的连接失败，配合启动自检的
   「DB_PASSWORD 未设置或为弱值」告警，比任何默认值都诚实；生产环境该键缺失本就是致命配置。
3. **模板只写键位与注释、不写任何真实值**，示例口令一律 `<openssl rand -hex 24>` 占位（《部署文档》§4.2 同步）。
4. **MONGO_URI 单独补检**：`deploy-selfcheck.js` 在容器内运行，拿不到 compose 插值前的
   `MONGO_ROOT_PASSWORD` 等变量，但 URI 内嵌的应用口令可以按特征直接识别。

**行为变更**：未设置 `DB_PASSWORD` 的本地环境连库将直接失败（原为静默回落 `123456`）。
本轮**无接口、无库表、无前端代码改动**；《部署文档》v6.11 同步 4 条运维注意。

**遗留**：①`.env` 实际口令的轮换属用户侧动作（本轮零值改动、未触碰任何现存口令，仅补键位与文档）；
②git 历史 d771d57/08c387b 中 `eudazi123` 曾作为 Mongo 应用账号口令进入版本库（I-05 时期已移除；
当前树内该串只存在于弱口令检测名单 `server.js:39-40`、`mongo/init.js` 历史注释与文档示例），历史未重写；
③`tests/` 与 `test_final.ps1` 内 `123456` 为测试夹具口令，不影响运行时；
④`WECHAT_APP_SECRET` 仅是代码回退别名（`authService.js` 读 `WECHAT_APPSECRET || WECHAT_APP_SECRET`），
模板只保留主键；⑤TURN 弱密钥目前仅启动告警、不阻断启动；⑥新增 `tests/unit/config/envHygiene.test.js`（4 例）

### 七、存储与带宽优化策略（2026-08-02 新增）

系统采用「前端直传 COS + 后端只存 URL 索引 + 本地兜底」的分层存储方案，目标：降低服务器带宽账单、保证内容不丢可审计。

1. **前端直传 COS**：前端先 `GET /api/upload/token` 拿预签名 PUT URL，直接把文件字节流上传到腾讯云 COS，绕过后端文件流中转，服务器只承担轻量预签名与 URL 登记请求。
2. **后端只存 URL 索引**：直传完成后 `POST /api/upload/register` 把访问 URL 登记到统一媒资表 `xn_media_asset`（含 user_id、biz_type、biz_id、storage、status），用于审计留痕与级联删除。服务器不保存文件本体。
3. **本地兜底**：COS 未配置或直传失败时，`frontend/src/services/uploadService.js` 自动回退到 `/api/upload/*` 后端中转（落盘 `public/uploads`，由 `uploadService.deleteFile` 管理）。
4. **访问签名 URL 防盗刷**：读取动态/相册媒体时，`cosSignedUrlService.getSignedAccessUrl` 生成带时效（默认 600s）的 COS GET 预签名 URL，避免公开永久直链被盗刷带宽。
5. **级联删除**：删除动态（`circleService.deletePost`）、删除相册照片（`albumService.deletePhoto`）、撤回聊天图片/视频/语音（`chatService.revokeMessage`）时，均调用 `mediaAssetService.removeByBiz` 同步清理底层存储（COS/本地）与媒资登记，避免孤儿文件。
6. **P2P 热内容取回流（可选加速）**：`/api/p2p/peer-online` 探测对方在线后，前端 `p2pFetchService` 通过 WebRTC DataChannel 让资源拥有方点对点回传小文件（头像/缩略图），进一步省服务器回源；不可达时回退到签名 URL。

> **上线前需手动配置**：① 腾讯云 COS 控制台开启「防盗链」referer 白名单（与本站域名一致）；② 配置 COS CORS 允许本站域名 PUT/GET；③ Nginx `/uploads/` 已配置 referer 校验（`valid_referers`），详见部署文档。

## 九、注意事项

1. **数据库连接池**: MySQL连接池默认最大10个连接，根据服务器配置调整
2. **索引优化**: 频繁查询的字段已添加索引，避免全表扫描
3. **事务处理**: 涉及多表操作时使用事务确保数据一致性
4. **日志记录**: 使用 `console.log` 记录日志到 `logs/` 目录
5. **安全建议**:
   - 生产环境务必修改JWT密钥
   - 敏感配置使用环境变量
   - 启用HTTPS
   - 定期备份数据库
   - `NODE_ENV` 必须显式设置（代码兜底值是 `development`；启动自检缺失时会告警）
   - 反向代理层级用 `TRUST_PROXY` 显式声明，默认只信任回环；绝不为图省事填 `true`（等于把 `req.ip` 交给请求方）
   - 限流不再按环境豁免，本地需要放行用 `RATE_LIMIT_DISABLED` / 各项 `*_RATE_LIMIT_MAX`（生产禁用前者）
6. **字段映射**: 用户信息返回字段为驼峰式（`userId`/`nickname`/`level`/`vipLevel`/`balance`/`giftBalance`/`fansCount`/`isFollow`/`description`/`sex`），注意昵称是 `nickname`（非 `nickName`），`sex` 保持数字（0-未知 1-男 2-女）
7. **货币单位**: 所有业务订单统一使用金币作为交易货币

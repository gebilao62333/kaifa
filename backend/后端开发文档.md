## 一、项目概述

### 1.1 项目简介

eu搭子平台后端是一个基于 Node.js + Express 的 RESTful API 服务，提供用户管理、聊天消息、礼物系统、订单管理、支付处理等核心功能。

### 1.2 技术栈

| 类别 | 技术 | 版本 |
|------|------|------|
| 运行时 | Node.js | >=16.0.0（package.json engines 声明） |
| 框架 | Express | 4.x |
| ORM（MySQL） | Sequelize | 6.x |
| ORM（MongoDB） | Mongoose | 8.x |
| 数据库 | MySQL | 8.0（mysql2 驱动） |
| 数据库 | MongoDB | —（Mongoose 8.x，配置保留于 `config/mongo.js`） |
| 缓存 | Redis | 4.x（redis）/ ioredis 6.x |
| 实时通讯 | Socket.IO | 4.x |
| 对象存储 | 腾讯云 COS | cos-nodejs-sdk-v5 3.x |
| 文档 | Swagger | swagger-jsdoc 6.x + swagger-ui-express 5.x |
| 认证 | JWT | jsonwebtoken 9.x |

| 测试 | Jest + Supertest | 29.x |
| 云服务 | 腾讯云 SDK | tencentcloud-sdk-nodejs 4.x |

#### JWT配置 (jwt.js)

> 注意：`jwt.js` 导出的是**工具函数**而非配置对象；密钥与过期时间配置在 `src/config/index.js` 的 `config.jwt` 项（secret / expiresIn / refreshExpiresIn）。

```javascript
// src/config/jwt.js
const jwt = require('jsonwebtoken')
const config = require('./index')

const generateToken = (payload, type = 'access') => {
  const expiresIn = type === 'access' ? config.jwt.expiresIn : config.jwt.refreshExpiresIn
  return jwt.sign(payload, config.jwt.secret, { expiresIn })
}

const generateTokenPair = (payload) => {
  return {
    accessToken: generateToken(payload, 'access'),
    refreshToken: generateToken(payload, 'refresh'),
    expiresIn: config.jwt.expiresIn,
    refreshExpiresIn: config.jwt.refreshExpiresIn
  }
}

const verifyToken = (token) => {
  try {
    return jwt.verify(token, config.jwt.secret)
  } catch (error) {
    return null // 校验失败返回 null，由调用方决定如何处理
  }
}

// 另导出 refreshAccessToken / decodeToken / signToken
module.exports = {
  generateToken,
  generateTokenPair,
  verifyToken,
  refreshAccessToken,
  decodeToken,
  signToken
}
```

### 2.2 中间件 (middlewares/)

#### 认证中间件 (auth.js)

```javascript
// src/middlewares/auth.js
const { verifyToken } = require('../config/jwt')
const { User } = require('../models')
const { getRedisClient } = require('../config/redis')

const authMiddleware = async (req, res, next) => {
  try {
    const token = req.headers.authorization?.replace('Bearer ', '')

    if (!token) {
      return res.status(401).json({ code: 401, message: '未提供认证令牌' })
    }

    // verifyToken 校验失败返回 null（不抛异常）
    const decoded = verifyToken(token)
    if (!decoded) {
      return res.status(401).json({ code: 401, message: '令牌无效或已过期' })
    }

    // Redis 黑名单检查（登出令牌作废），2 秒兜底防止 Redis 故障卡死认证
    const redis = getRedisClient()
    if (redis) {
      const isBlacklisted = await Promise.race([
        redis.get(`blacklist:${token}`),
        new Promise((resolve) => setTimeout(() => resolve(null), 2000))
      ]).catch(() => null)
      if (isBlacklisted) {
        return res.status(401).json({ code: 401, message: '令牌已失效' })
      }
    }

    const user = await User.findByPk(decoded.userId)
    if (!user) {
      return res.status(401).json({ code: 401, message: '用户不存在' })
    }

    // status: 1=正常，其余=禁用
    if (user.status !== 1) {
      return res.status(403).json({ code: 403, message: '账号已被禁用' })
    }

    // jinyan_time: 禁言截止时间戳
    const now = Math.floor(Date.now() / 1000)
    if (user.jinyan_time && user.jinyan_time > now) {
      return res.status(403).json({ code: 403, message: '用户已被禁言' })
    }

    req.user = user
    req.userId = decoded.userId
    req.token = token
    next()
  } catch (error) {
    logger.error('认证中间件错误:', error)
    return res.status(500).json({ code: 500, message: '服务器内部错误' })
  }
}

// 可选认证：无 token 或 token 无效时直接放行，有效时挂载 req.user/req.userId
const optionalAuth = async (req, res, next) => { /* 同上，省略 */ }

// 管理员认证：支持 Authorization: Bearer 或 x-admin-token 两种传递方式
// 1. ADMIN_TOKEN 环境变量应急令牌（config.admin.token）
// 2. 管理员登录签发的 JWT（decoded.role === 'admin' 或 decoded.role_id === 1）
const adminAuth = (req, res, next) => { /* ... */ }

module.exports = { authMiddleware, optionalAuth, adminAuth }
```

#### 限流中间件 (rateLimit.js)

```javascript
// src/middlewares/rateLimit.js
const rateLimit = require('express-rate-limit')
const config = require('../config')

// 开发环境跳过限流，避免调试干扰
const skipInDev = (req, res) => config.nodeEnv === 'development'

// 全局限流：窗口与阈值取自 config.rateLimit（config/index.js）
const apiLimiter = rateLimit({
  windowMs: config.rateLimit.windowMs,
  max: config.rateLimit.maxRequests,
  message: { code: 429, message: '请求过于频繁，请稍后再试' },
  standardHeaders: true,
  legacyHeaders: false,
  skip: skipInDev
})

// 登录限流：15分钟内最多 100 次
const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  message: { code: 429, message: '登录尝试次数过多，请15分钟后再试' },

})

// 短信验证码限流：60秒内最多 1 次
```

### 2.3 控制器层 (controllers/)

> **编码风格**：控制器是**薄层委托**——只做参数提取与校验，业务逻辑全部委托给 `services/`，响应统一通过 `src/utils/response.js` 工具输出（`response.success / created / badRequest / notFound / unprocessableEntity / error` 等），**不存在** `res.ok()` / `res.fail()` 这类方法。

#### 用户控制器示例

```javascript
// src/controllers/user.js
const { authService, smsService } = require('../services')
const response = require('../utils/response')
const logger = require('../utils/logger')

const login = async (req, res) => {
  try {
    const { username, mobile, phone, password } = req.body
    const account = username || mobile || phone

    if (!account || !password) {
      return response.badRequest(res, '手机号和密码不能为空')
    }

    const result = await authService.loginWithPassword(account, password)
    response.success(res, result, '登录成功')
  } catch (error) {
    logger.error('登录错误:', error)
    if (error.message === '用户不存在') {
      response.unprocessableEntity(res, '用户不存在', { phone: ['用户不存在，请检查手机号或使用验证码登录'] })
    } else if (error.message === '密码错误') {
      response.unprocessableEntity(res, '密码错误', { password: ['密码错误，请重新输入'] })
    } else {
      response.error(res, '登录失败')
    }
  }
}

// 获取用户信息：薄层委托 authService，支持查看他人主页（?userId=）
const getUserInfo = async (req, res) => {
  try {
    const targetUserId = req.query.userId || req.userId
    const userInfo = await authService.getUserInfo(req.userId, targetUserId)
    response.success(res, userInfo)
  } catch (error) {
    logger.error('获取用户信息错误:', error)
    response.error(res, error.message)
  }
}

// authService.getUserInfo 返回字段：
// { userId, nickname, avatar, city, level, vip, vipLevel, balance, giftBalance,
//   score, fansCount, isFollow, description, sex }
// 注意：昵称字段是 nickname（非 nickName），sex 保持数字（0-未知 1-男 2-女）

// 更新用户信息
const updateUserInfo = async (req, res) => {
  try {
    await authService.updateUserInfo(req.userId, req.body)
    response.success(res, {}, '更新成功')
  } catch (error) {
    logger.error('更新用户信息错误:', error)
    response.error(res, error.message)
  }
}

// 关注/取消关注：同一个 /follow 端点，通过 body.action 区分（无独立 /unfollow 端点）
// 关注关系存于 xn_user_follow 表，列名为 follower_id / following_id
const follow = async (req, res) => { /* 委托 authService，省略 */ }

// 其余方法：register / resetPassword / sendSms / loginMobile / loginThird /
// getFans / getFollows / checkFollow / refreshToken
module.exports = { login, register, resetPassword, getUserInfo, updateUserInfo, follow, getFans, getFollows, checkFollow, sendSms, loginMobile, loginThird, refreshToken }
```

#### 聊天控制器示例

> **注意**：聊天消息当前走 **MySQL**（`chatService` 使用 `ChatSession` / `ChatLog` / `VirtualUser` 模型），MongoDB 侧用于其他高频写入场景的扩展存储。

```javascript
// src/controllers/chat.js
const { chatService } = require('../services')
const response = require('../utils/response')
const logger = require('../utils/logger')

// 获取聊天列表（会话存于 MySQL xn_chat_room 表，对端为虚拟用户）
const getChatList = async (req, res) => {
  try {
    const { page = 1, pageSize = 20 } = req.query
    const result = await chatService.getChatList(req.userId, parseInt(page), parseInt(pageSize))
    response.success(res, result)
  } catch (error) {
    logger.error('获取聊天列表错误:', error)
    response.error(res, error.message)
  }
}

// 获取消息列表（ChatLog 列：fromid/toid/content/type/vod_url/sec/time/isread/is_del/is_revoked）
const getMessages = async (req, res) => {
  try {
    const { targetUserId, page = 1, pageSize = 20 } = req.query
    if (!targetUserId) {
      return response.badRequest(res, '目标用户ID不能为空')
    }
    const result = await chatService.getChatMessages(
      req.userId, parseInt(targetUserId), parseInt(page), parseInt(pageSize)
    )
    response.success(res, result)
  } catch (error) {
    logger.error('获取聊天消息错误:', error)
    response.error(res, error.message)
  }
}

// 发送消息（写入 MySQL xn_chat_log，并通过 Socket.IO 推送给对端）
const sendMessage = async (req, res) => {
  try {
    const { targetUserId, content, type = 0, mediaUrl, duration } = req.body
    if (!targetUserId || !content) {
      return response.badRequest(res, '接收者ID和内容不能为空')
    }
    const result = await chatService.sendMessage(
      req.userId, parseInt(targetUserId), content, parseInt(type),
      mediaUrl, duration ? parseInt(duration) : undefined
    )
    response.success(res, result, '发送成功')
  } catch (error) {
    logger.error('发送消息错误:', error)
    response.unprocessableEntity(res, error.message)
  }
}

// 撤回消息（is_revoked 置 1，媒体消息同步清理 COS/本地存储与媒资登记）
const revokeMessage = async (req, res) => { /* 委托 chatService.revokeMessage，省略 */ }

// 其余方法：createRoom / getRoomInfo / getRooms / markAsRead
module.exports = { getChatList, getMessages, sendMessage, revokeMessage, createRoom, getRoomInfo, getRooms, markAsRead }
```

```javascript
const { ChatLog, ChatRoom, User, VirtualUser, ChatSession } = require('../models')

// 聊天列表：会话存于 MySQL xn_chat_room（ChatSession 模型，user_id <-> virtual_user_id）
const getChatList = async (userId, page, pageSize) => {
  const { offset, limit } = parseQuery({ page, pageSize })

  const { count, rows } = await ChatSession.findAndCountAll({
    where: { user_id: userId },
    offset,
    limit,
    order: [['update_time', 'DESC']]
  })

  const list = await Promise.all(rows.map(async (session) => {
    const peerUser = await VirtualUser.findByPk(session.virtual_user_id, {
      attributes: ['id', 'name', 'avatar', 'status']
    })
    return {
      id: session.id,
      fromId: session.virtual_user_id,
      toId: userId,
      nickname: peerUser?.name || '',
      avatar: peerUser?.avatar || '',
      content: session.last_message,
      sendTime: session.last_message_time,
      unreadCount: session.unread_count
    }
  }))

  return { total: count, list }
}

// 消息历史：ChatLog 列 fromid/toid/content/type/vod_url/sec/time/is_del/is_revoked
const getChatMessages = async (userId, targetUserId, page, pageSize) => {
  const { offset, limit } = parseQuery({ page, pageSize })

  const { count, rows } = await ChatLog.findAndCountAll({
    where: {
      [Op.or]: [
        { fromid: userId, toid: targetUserId },
        { fromid: targetUserId, toid: userId }
      ],
      is_del: 0
    },
    offset,
    limit,
    order: [['time', 'DESC']]
  })

  const messages = rows.map(msg => ({
    id: msg.id,
    fromId: msg.fromid,
    toId: msg.toid,
    content: msg.content,
    type: msg.type,
    mediaUrl: msg.vod_url,
    duration: msg.sec,
    sendTime: msg.time,
    isSelf: msg.fromid === userId,
    isRevoked: msg.is_revoked === 1
  }))

  return { total: count, list: messages.reverse() }
}
```

#### 通话计费服务 (callBillingService.js)

```javascript
// src/services/callBillingService.js
class CallBillingService {
  /**
   * 创建通话记录
   * @param {Object} params - 通话参数
   * @param {number} params.callerId - 主叫用户ID
   * @param {number} params.calleeId - 被叫用户ID
   * @param {number} params.type - 通话类型 1-语音 2-视频
   * @param {number} params.pricePerMin - 每分钟价格（金币）
   */
  async createCallRecord({ callerId, calleeId, type, pricePerMin }) {
    return await CallRecord.create({
      caller_id: callerId,
      callee_id: calleeId,
      type,
      price_per_min: pricePerMin,
      start_time: Date.now(),
      status: 1
    })
  }

  /**
   * 结束通话并结算费用
   * @param {number} recordId - 通话记录ID
   * @param {number} duration - 通话时长（秒）
   */
  async endCall(recordId, duration) {
    const record = await CallRecord.findByPk(recordId)
    const minutes = Math.ceil(duration / 60)
    const totalCost = minutes * record.price_per_min

    // 扣除主叫方余额
    await User.decrement('money', {
      by: totalCost,
      where: { id: record.caller_id }
    })

    // 增加被叫方收入
    await User.increment('money', {
      by: totalCost * 0.8, // 平台抽成20%
      where: { id: record.callee_id }
    })

    // 更新通话记录状态
    await record.update({
      duration,
      total_cost: totalCost,
      end_time: Date.now(),
      status: 2
    })

    // 记录计费明细
    await CallBilling.create({
      record_id: recordId,
      duration,
      unit_price: record.price_per_min,
      total_amount: totalCost,
      caller_id: record.caller_id,
      callee_id: record.callee_id
    })

    return totalCost
  }
}

module.exports = new CallBillingService()
```

### 七、存储与带宽优化策略（2026-08-02 新增）

系统采用「前端直传 COS + 后端只存 URL 索引 + 本地兜底」的分层存储方案，目标：降低服务器带宽账单、保证内容不丢可审计。

1. **前端直传 COS**：前端先 `GET /api/upload/token` 拿预签名 PUT URL，直接把文件字节流上传到腾讯云 COS，绕过后端文件流中转，服务器只承担轻量预签名与 URL 登记请求。
2. **后端只存 URL 索引**：直传完成后 `POST /api/upload/register` 把访问 URL 登记到统一媒资表 `xn_media_asset`（含 user_id、biz_type、biz_id、storage、status），用于审计留痕与级联删除。服务器不保存文件本体。
3. **本地兜底**：COS 未配置或直传失败时，`frontend/src/services/uploadService.js` 自动回退到 `/api/upload/*` 后端中转（落盘 `public/uploads`，由 `uploadService.deleteFile` 管理）。
4. **访问签名 URL 防盗刷**：读取动态/相册媒体时，`cosSignedUrlService.getSignedAccessUrl` 生成带时效（默认 600s）的 COS GET 预签名 URL，避免公开永久直链被盗刷带宽。
5. **级联删除**：删除动态（`circleService.deletePost`）、删除相册照片（`albumService.deletePhoto`）、撤回聊天图片/视频/语音（`chatService.revokeMessage`）时，均调用 `mediaAssetService.removeByBiz` 同步清理底层存储（COS/本地）与媒资登记，避免孤儿文件。
6. **P2P 热内容取回流（可选加速）**：`/api/p2p/peer-online` 探测对方在线后，前端 `p2pFetchService` 通过 WebRTC DataChannel 让资源拥有方点对点回传小文件（头像/缩略图），进一步省服务器回源；不可达时回退到签名 URL。

> **上线前需手动配置**：① 腾讯云 COS 控制台开启「防盗链」referer 白名单（与本站域名一致）；② 配置 COS CORS 允许本站域名 PUT/GET；③ Nginx `/uploads/` 已配置 referer 校验（`valid_referers`），详见部署文档。

## 九、注意事项

1. **数据库连接池**: MySQL连接池默认最大10个连接，根据服务器配置调整
2. **索引优化**: 频繁查询的字段已添加索引，避免全表扫描
3. **事务处理**: 涉及多表操作时使用事务确保数据一致性
4. **日志记录**: 使用 `console.log` 记录日志到 `logs/` 目录
5. **安全建议**:
   - 生产环境务必修改JWT密钥
   - 敏感配置使用环境变量
   - 启用HTTPS
   - 定期备份数据库
6. **字段映射**: 用户信息返回字段为驼峰式（`userId`/`nickname`/`level`/`vipLevel`/`balance`/`giftBalance`/`fansCount`/`isFollow`/`description`/`sex`），注意昵称是 `nickname`（非 `nickName`），`sex` 保持数字（0-未知 1-男 2-女）
7. **货币单位**: 所有业务订单统一使用金币作为交易货币
